cmake_minimum_required(VERSION 3.8)
project(senior_motion_control)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(senior_motion_msgs 0.3.0 REQUIRED)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/safety_node scripts/command_node DESTINATION lib/${PROJECT_NAME})
add_executable(final_command_guard native/final_command_guard.cpp)
ament_target_dependencies(final_command_guard rclcpp senior_motion_msgs)
add_executable(motion_output_guard native/motion_output_guard.cpp)
ament_target_dependencies(motion_output_guard rclcpp geometry_msgs senior_motion_msgs)
target_compile_features(final_command_guard PRIVATE cxx_std_17)
target_compile_features(motion_output_guard PRIVATE cxx_std_17)
install(TARGETS final_command_guard motion_output_guard DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_test(NAME final_command_frame_roundtrip
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_final_command_frame.py)
  set_tests_properties(final_command_frame_roundtrip PROPERTIES
    ENVIRONMENT "CXX=${CMAKE_CXX_COMPILER}")
endif()
ament_package()
