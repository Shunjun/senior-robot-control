cmake_minimum_required(VERSION 3.8)
project(senior_scan_matcher)
find_package(ament_cmake REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(rclcpp REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(senior_motion_msgs REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread)
find_package(TBB REQUIRED)
find_library(KARTO_LIBRARY NAMES kartoSlamToolbox REQUIRED)
add_library(scan_matcher src/scan_matcher.cpp src/match_diagnostics.cpp)
target_compile_features(scan_matcher PUBLIC cxx_std_17)
target_include_directories(scan_matcher PUBLIC include)
ament_target_dependencies(scan_matcher slam_toolbox)
target_link_libraries(scan_matcher ${KARTO_LIBRARY} Boost::serialization Boost::thread TBB::tbb)
add_executable(scan_motion_node src/scan_motion_node.cpp)
target_compile_features(scan_motion_node PUBLIC cxx_std_17)
target_link_libraries(scan_motion_node scan_matcher)
ament_target_dependencies(scan_motion_node diagnostic_msgs rclcpp sensor_msgs geometry_msgs tf2_ros senior_motion_msgs)
add_executable(scan_matcher_replay src/scan_matcher_replay.cpp)
target_compile_definitions(scan_matcher_replay PRIVATE MATCH_DIAGNOSTICS)
target_link_libraries(scan_matcher_replay scan_matcher)
install(TARGETS scan_matcher_replay DESTINATION lib/${PROJECT_NAME})
install(TARGETS scan_motion_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  add_test(NAME replay_stdout_contract COMMAND /usr/bin/python3
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_replay_cli.py $<TARGET_FILE:scan_matcher_replay>)
  ament_add_gtest(test_scan_motion_shutdown test/test_scan_motion_shutdown.cpp)
  target_compile_features(test_scan_motion_shutdown PUBLIC cxx_std_17)
  target_link_libraries(test_scan_motion_shutdown scan_matcher)
  ament_target_dependencies(test_scan_motion_shutdown diagnostic_msgs rclcpp sensor_msgs geometry_msgs tf2_ros senior_motion_msgs)
  ament_add_gtest(test_diagnostic_isolation test/test_diagnostic_isolation.cpp)
  target_link_libraries(test_diagnostic_isolation scan_matcher)
  ament_target_dependencies(test_diagnostic_isolation diagnostic_msgs senior_motion_msgs)
  ament_add_gtest(test_match_diagnostics test/test_match_diagnostics.cpp)
  target_link_libraries(test_match_diagnostics scan_matcher)
  ament_add_gtest(test_scan_matcher test/test_scan_matcher.cpp)
  target_link_libraries(test_scan_matcher scan_matcher)
endif()
ament_package()
