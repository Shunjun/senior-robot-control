cmake_minimum_required(VERSION 3.8)
project(senior_navigation)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(senior_interfaces REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(nlohmann_json 3.10.5 REQUIRED)

add_library(senior_current_scan_layer SHARED src/current_scan_layer.cpp)
target_compile_features(senior_current_scan_layer PUBLIC cxx_std_17)
target_link_libraries(senior_current_scan_layer nlohmann_json::nlohmann_json)
target_include_directories(senior_current_scan_layer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(senior_current_scan_layer
  nav2_costmap_2d geometry_msgs pluginlib rclcpp rcl_interfaces senior_interfaces tf2 tf2_ros tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d current_scan_layer.xml)
install(TARGETS senior_current_scan_layer EXPORT senior_current_scan_layer
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(senior_current_scan_layer HAS_LIBRARY_TARGET)
ament_export_dependencies(nav2_costmap_2d geometry_msgs senior_interfaces nlohmann_json)

ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/navigation_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config behavior_trees DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(nav2_util REQUIRED)
  find_package(nav2_smac_planner REQUIRED)
  find_package(nav2_mppi_controller REQUIRED)
  ament_add_gtest(test_current_scan_layer test/test_current_scan_layer.cpp)
  target_link_libraries(test_current_scan_layer senior_current_scan_layer)
  ament_target_dependencies(test_current_scan_layer
    nav2_costmap_2d nav2_util nav2_smac_planner nav2_mppi_controller)
endif()
ament_package()
