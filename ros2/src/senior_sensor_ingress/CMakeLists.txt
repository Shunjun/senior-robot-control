cmake_minimum_required(VERSION 3.8)
project(senior_sensor_ingress)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(senior_motion_msgs REQUIRED)
add_library(raw_receipt src/raw_receipt.cpp)
target_compile_features(raw_receipt PUBLIC cxx_std_17)
target_include_directories(raw_receipt PUBLIC include)
ament_target_dependencies(raw_receipt rclcpp sensor_msgs std_msgs senior_motion_msgs)
add_library(scan_receipt src/scan_receipt.cpp)
target_compile_features(scan_receipt PUBLIC cxx_std_17)
target_include_directories(scan_receipt PUBLIC include)
target_link_libraries(scan_receipt raw_receipt)
ament_target_dependencies(scan_receipt rclcpp sensor_msgs senior_motion_msgs)
add_executable(scan_receipt_node src/scan_main.cpp)
target_link_libraries(scan_receipt_node scan_receipt)
ament_target_dependencies(scan_receipt_node rclcpp)
add_executable(raw_receipt_node src/main.cpp)
target_link_libraries(raw_receipt_node raw_receipt)
ament_target_dependencies(raw_receipt_node rclcpp)
install(TARGETS raw_receipt raw_receipt_node scan_receipt scan_receipt_node
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_scan_egress test/test_scan_egress.cpp)
  target_link_libraries(test_scan_egress scan_receipt)
  ament_target_dependencies(test_scan_egress rclcpp sensor_msgs senior_motion_msgs)
  ament_add_gtest(test_raw_scan_receipt test/test_raw_scan_receipt.cpp)
  target_link_libraries(test_raw_scan_receipt scan_receipt)
  ament_target_dependencies(test_raw_scan_receipt rclcpp sensor_msgs senior_motion_msgs)
  ament_add_gtest(test_scan_ledger test/test_scan_ledger.cpp)
  target_include_directories(test_scan_ledger PRIVATE include)
  ament_target_dependencies(test_scan_ledger senior_motion_msgs)
  ament_add_gtest(test_capture_ledger test/test_capture_ledger.cpp)
  target_include_directories(test_capture_ledger PRIVATE include)
  ament_target_dependencies(test_capture_ledger senior_motion_msgs)
  ament_add_gtest(test_semantic_egress test/test_semantic_egress.cpp)
  target_link_libraries(test_semantic_egress raw_receipt)
  ament_target_dependencies(test_semantic_egress rclcpp sensor_msgs senior_motion_msgs)
  ament_add_gtest(test_depth_egress test/test_depth_egress.cpp)
  target_link_libraries(test_depth_egress raw_receipt)
  ament_target_dependencies(test_depth_egress rclcpp sensor_msgs senior_motion_msgs)
  ament_add_gtest(test_raw_receipt test/test_raw_receipt.cpp)
  target_link_libraries(test_raw_receipt raw_receipt)
  ament_target_dependencies(test_raw_receipt rclcpp sensor_msgs std_msgs senior_motion_msgs)
endif()
ament_package()
