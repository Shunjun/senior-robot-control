cmake_minimum_required(VERSION 3.8)
project(senior_sensor_ingress)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../tools/ros_wiring.cmake")
set(wiring_include "${CMAKE_CURRENT_BINARY_DIR}/generated")
senior_generate_wiring("${wiring_include}/senior_sensor_ingress/ros_wiring.hpp")
include_directories("${wiring_include}")
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
find_package(nav_msgs REQUIRED)

# Native MessageInfo receipt uses the message package's installed C++ export.
add_executable(native_receipt src/native_receipt.cpp)
target_compile_features(native_receipt PRIVATE cxx_std_17)
target_include_directories(native_receipt PRIVATE include)
ament_target_dependencies(native_receipt rclcpp nav_msgs sensor_msgs geometry_msgs std_msgs senior_interfaces)
install(TARGETS native_receipt DESTINATION lib/${PROJECT_NAME})
add_library(rgbd_candidate_ledger src/rgbd_candidate_ledger.cpp)
target_compile_features(rgbd_candidate_ledger PUBLIC cxx_std_17)
target_include_directories(rgbd_candidate_ledger PUBLIC include)
target_link_libraries(rgbd_candidate_ledger OpenSSL::Crypto)
ament_target_dependencies(rgbd_candidate_ledger senior_interfaces)
add_library(rgbd_candidate_receipt src/rgbd_candidate_receipt.cpp)
target_compile_features(rgbd_candidate_receipt PUBLIC cxx_std_17)
target_include_directories(rgbd_candidate_receipt PUBLIC include)
target_link_libraries(rgbd_candidate_receipt rgbd_candidate_ledger raw_receipt)
ament_target_dependencies(rgbd_candidate_receipt rclcpp tf2_ros tf2_msgs geometry_msgs senior_interfaces)
add_executable(rgbd_candidate_receipt_node src/rgbd_candidate_main.cpp)
target_link_libraries(rgbd_candidate_receipt_node rgbd_candidate_receipt)
ament_target_dependencies(rgbd_candidate_receipt_node rclcpp)
add_library(raw_receipt src/raw_receipt.cpp)
target_compile_features(raw_receipt PUBLIC cxx_std_17)
target_include_directories(raw_receipt PUBLIC include)
ament_target_dependencies(raw_receipt rclcpp sensor_msgs std_msgs senior_interfaces)
add_library(scan_receipt src/scan_receipt.cpp)
target_compile_features(scan_receipt PUBLIC cxx_std_17)
target_include_directories(scan_receipt PUBLIC include)
target_link_libraries(scan_receipt raw_receipt)
ament_target_dependencies(scan_receipt rclcpp sensor_msgs senior_interfaces)
add_executable(scan_receipt_node src/scan_main.cpp)
target_link_libraries(scan_receipt_node scan_receipt)
ament_target_dependencies(scan_receipt_node rclcpp)
add_executable(raw_receipt_node src/main.cpp)
target_link_libraries(raw_receipt_node raw_receipt)
ament_target_dependencies(raw_receipt_node rclcpp)
install(TARGETS raw_receipt raw_receipt_node scan_receipt scan_receipt_node rgbd_candidate_ledger rgbd_candidate_receipt rgbd_candidate_receipt_node
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})

# Public monotonic capture contracts, independent of generated message targets.
add_library(sensor_timing INTERFACE)
target_include_directories(sensor_timing INTERFACE
  "$<BUILD_INTERFACE:${wiring_include}>"
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
  "$<INSTALL_INTERFACE:include/${PROJECT_NAME}>")
target_compile_features(sensor_timing INTERFACE cxx_std_17)
install(DIRECTORY include/ DESTINATION include/${PROJECT_NAME})
install(DIRECTORY "${wiring_include}/" DESTINATION include/${PROJECT_NAME})
install(TARGETS sensor_timing EXPORT export_sensor_timing)
ament_export_targets(export_sensor_timing HAS_LIBRARY_TARGET)
