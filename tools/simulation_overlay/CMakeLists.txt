# Three deployed systems use the reset-aware fixed core built by core.CMakeLists.txt.
cmake_minimum_required(VERSION 3.16)
project(senior_simulation_systems LANGUAGES CXX)
find_package(ignition-gazebo6 6.17.0 EXACT REQUIRED COMPONENTS rendering)
find_package(ignition-sensors6 6.8.1 EXACT REQUIRED COMPONENTS
  camera depth_camera gpu_lidar lidar rgbd_camera boundingbox_camera segmentation_camera thermal_camera imu)
find_package(ignition-rendering6 REQUIRED)
find_package(ignition-common4 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
find_package(senior_motion_msgs 0.2.0 EXACT REQUIRED)
if(NOT DEFINED GZ_SIM_SOURCE)
  message(FATAL_ERROR "GZ_SIM_SOURCE must name the verified selected source directory")
endif()
foreach(system IN ITEMS sensors imu odometry_publisher)
  if(system STREQUAL "sensors")
    set(source Sensors)
    set(library ignition-gazebo6-sensors-system)
  elseif(system STREQUAL "imu")
    set(source Imu)
    set(library ignition-gazebo6-imu-system)
  else()
    set(source OdometryPublisher)
    set(library ignition-gazebo6-odometry-publisher-system)
  endif()
  add_library(${library} SHARED "${GZ_SIM_SOURCE}/src/systems/${system}/${source}.cc")
  string(REPLACE "gazebo6-" "gazebo-" runtime_name "${library}")
  set_target_properties(${library} PROPERTIES OUTPUT_NAME "${runtime_name}" INSTALL_RPATH "$ORIGIN")
  target_compile_features(${library} PUBLIC cxx_std_17)
  target_link_libraries(${library} PRIVATE ignition-gazebo6::core
    ignition-common4::ignition-common4 ignition-plugin1::register
    senior_motion_msgs::sensor_timing senior_motion_msgs::simulation_timing)
  if(system STREQUAL "sensors")
    target_link_libraries(${library} PRIVATE ignition-gazebo6::rendering
      ignition-rendering6::ignition-rendering6)
    foreach(component IN ITEMS camera depth_camera gpu_lidar lidar rgbd_camera
        boundingbox_camera segmentation_camera thermal_camera)
      target_link_libraries(${library} PRIVATE ignition-sensors6::${component})
    endforeach()
  elseif(system STREQUAL "imu")
    target_link_libraries(${library} PRIVATE ignition-sensors6::imu)
  endif()
  install(TARGETS ${library} LIBRARY DESTINATION lib)
endforeach()
