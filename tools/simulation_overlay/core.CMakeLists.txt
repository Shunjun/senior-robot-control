# Build only the fixed upstream core source list, against the exact installed SDK.
# GUI, tools, examples, tests and unrelated systems are not copied or rebuilt.
cmake_minimum_required(VERSION 3.16)
project(senior_gazebo_core LANGUAGES CXX)
find_package(ignition-gazebo6 6.17.0 EXACT REQUIRED)
find_package(ignition-msgs8 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS loader)
find_package(Protobuf REQUIRED)
find_package(senior_motion_msgs 0.2.0 EXACT REQUIRED)
get_target_property(epoch_library_type senior_motion_msgs::simulation_timing TYPE)
if(NOT epoch_library_type STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "World epoch state must live in one SHARED library")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/epoch-library-path.txt"
  CONTENT "$<TARGET_FILE:senior_motion_msgs::simulation_timing>\n")
if(NOT DEFINED GZ_SIM_SOURCE)
  message(FATAL_ERROR "GZ_SIM_SOURCE must name the verified fixed source closure")
endif()
get_target_property(sdk_includes ignition-gazebo6::core INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(sdk_links ignition-gazebo6::core INTERFACE_LINK_LIBRARIES)
get_target_property(msgs_includes ignition-msgs8::ignition-msgs8 INTERFACE_INCLUDE_DIRECTORIES)
find_path(msgs_proto_root ignition/msgs/header.proto HINTS ${msgs_includes})
if(NOT msgs_proto_root)
  message(FATAL_ERROR "The exact ignition-msgs8 SDK must supply its protobuf imports")
endif()
set(PROTOBUF_IMPORT_DIRS "${msgs_proto_root}")
# Keep upstream generation in its msgs binary subdirectory: source headers
# include msgs/*.pb.h, not flat private protobuf header names.
add_subdirectory("${GZ_SIM_SOURCE}/src/msgs" "${CMAKE_CURRENT_BINARY_DIR}/msgs")
set(core_sources
  "${GZ_SIM_SOURCE}/src/Actor.cc"
  "${GZ_SIM_SOURCE}/src/Barrier.cc"
  "${GZ_SIM_SOURCE}/src/BaseView.cc"
  "${GZ_SIM_SOURCE}/src/ComponentFactory.cc"
  "${GZ_SIM_SOURCE}/src/Conversions.cc"
  "${GZ_SIM_SOURCE}/src/EntityComponentManager.cc"
  "${GZ_SIM_SOURCE}/src/Joint.cc"
  "${GZ_SIM_SOURCE}/src/LevelManager.cc"
  "${GZ_SIM_SOURCE}/src/Light.cc"
  "${GZ_SIM_SOURCE}/src/Link.cc"
  "${GZ_SIM_SOURCE}/src/Model.cc"
  "${GZ_SIM_SOURCE}/src/Primitives.cc"
  "${GZ_SIM_SOURCE}/src/SdfEntityCreator.cc"
  "${GZ_SIM_SOURCE}/src/SdfGenerator.cc"
  "${GZ_SIM_SOURCE}/src/Sensor.cc"
  "${GZ_SIM_SOURCE}/src/Server.cc"
  "${GZ_SIM_SOURCE}/src/ServerConfig.cc"
  "${GZ_SIM_SOURCE}/src/ServerPrivate.cc"
  "${GZ_SIM_SOURCE}/src/SimulationRunner.cc"
  "${GZ_SIM_SOURCE}/src/SystemLoader.cc"
  "${GZ_SIM_SOURCE}/src/SystemManager.cc"
  "${GZ_SIM_SOURCE}/src/TestFixture.cc"
  "${GZ_SIM_SOURCE}/src/Util.cc"
  "${GZ_SIM_SOURCE}/src/View.cc"
  "${GZ_SIM_SOURCE}/src/World.cc"
  "${GZ_SIM_SOURCE}/src/comms/Broker.cc"
  "${GZ_SIM_SOURCE}/src/comms/ICommsModel.cc"
  "${GZ_SIM_SOURCE}/src/comms/MsgManager.cc"
  "${GZ_SIM_SOURCE}/src/network/NetworkConfig.cc"
  "${GZ_SIM_SOURCE}/src/network/NetworkManager.cc"
  "${GZ_SIM_SOURCE}/src/network/NetworkManagerPrimary.cc"
  "${GZ_SIM_SOURCE}/src/network/NetworkManagerSecondary.cc"
  "${GZ_SIM_SOURCE}/src/network/PeerInfo.cc"
  "${GZ_SIM_SOURCE}/src/network/PeerTracker.cc"
)
add_library(senior_gazebo_core SHARED ${core_sources} ${PROTO_PRIVATE_SRC})
set_target_properties(senior_gazebo_core PROPERTIES OUTPUT_NAME ignition-gazebo6
  VERSION 6.17.0 SOVERSION 6 INSTALL_RPATH "$ORIGIN")
target_compile_features(senior_gazebo_core PUBLIC cxx_std_17)
add_dependencies(senior_gazebo_core ignition-gazebo_private_msgs)
target_include_directories(senior_gazebo_core PRIVATE ${sdk_includes}
  "${GZ_SIM_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}")
# Reuse the SDK dependency metadata, never link the old core into its replacement.
if("ignition-gazebo6::core" IN_LIST sdk_links)
  message(FATAL_ERROR "Unexpected self dependency in fixed Gazebo core SDK metadata")
endif()
target_link_libraries(senior_gazebo_core PRIVATE ${sdk_links}
  ignition-plugin1::loader protobuf::libprotobuf senior_motion_msgs::simulation_timing)
if(UNIX AND NOT APPLE)
  target_link_libraries(senior_gazebo_core PRIVATE stdc++fs)
endif()
install(TARGETS senior_gazebo_core LIBRARY DESTINATION lib)
